
// Added only to the staged probe copy to render the actual private menu components.
@MainActor func rangedProbeMenuRow(layout: TowerMenuLayout, side: CGFloat) -> some View {
    HStack(spacing: 12) {
        TowerMenuItem(towerMenuLayout: layout, kind: .ranged, isAvailable: true,
            isArmed: false, isAffordable: true, cost: 70, buttonSize: CGSize(width: side, height: side)) {}
        ForEach([TowerKind.melee, .areaOfEffect, .special]) { kind in
            TowerMenuItem(towerMenuLayout: layout, kind: kind, isAvailable: true,
                isArmed: false, isAffordable: true, cost: 100, buttonSize: CGSize(width: side, height: side)) {}
        }
        ForEach(1...3, id: \.self) { branch in
            UpgradeMenuItem(towerMenuLayout: layout,
                iconName: TowerKind.ranged.assetName(atLevel: 4, branch: branch)!, dropKind: .ranged,
                cost: 300, isArmed: false, isAffordable: true, buttonSize: CGSize(width: side, height: side)) {}
        }
        TowerMenuItem(towerMenuLayout: layout, kind: .ranged, isAvailable: true,
            isArmed: false, isAffordable: false, cost: 70, buttonSize: CGSize(width: side, height: side)) {}
        TowerMenuItem(towerMenuLayout: layout, kind: .ranged, isAvailable: true,
            isArmed: true, isAffordable: true, cost: 70, buttonSize: CGSize(width: side, height: side)) {}
    }.padding(12).padding(.bottom, 10).background(Color(red: 0.2, green: 0.3, blue: 0.2))
}
